def calculate_sombor_indices(edge_data):
    """
    edge_data: list of tuples (d_u, d_v, count)
    Implements the seven Sombor indices exactly as provided.
    """
    SO = 0
    SO3 = 0
    SO4 = 0
    SO5 = 0
    SO6 = 0
    ESO = 0

    for du, dv, count in edge_data:
        term = math.sqrt(du**2 + dv**2)

        # (1) Standard Sombor
        SO += count * term

        # (2) SO3
        SO3 += count * (math.sqrt(2) * math.pi * ((du**2 + dv**2) / (du + dv)))

        # (3) SO4
        SO4 += count * ((math.pi / 2) * (((du**2 + dv**2) / (du + dv)) ** 2))

        # (4) SO5 (absolute value to ensure non-negative)
        denom = (math.sqrt(2) + 2 * term)
        SO5 += count * (2 * math.pi * (abs(du**2 - dv**2) / denom))

        # (5) SO6 (same rule, use absolute value)
        SO6 += count * (math.pi * ((abs(du**2 - dv**2) / denom) ** 2))

        # (6) ESO(G)
        ESO += count * ((du + dv) * term)

    # (7) MESO(G)
    MESO = 1 / ESO if ESO != 0 else 0

    return SO, SO3, SO4, SO5, SO6, ESO, MESO
